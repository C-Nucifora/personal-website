# Blog Window Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `6:blog` window fed by `data/blog/*.md`, dormant until the first post — per `docs/superpowers/specs/2026-06-12-blog-window-design.md`.

**Architecture:** `scripts/bundle-blog.mjs` parses markdown+frontmatter into `data/generated/blog.ts`. `WINDOW_IDS` gains `"blog"` (type source); a new `ACTIVE_WINDOW_IDS` (drops blog at zero posts) drives every UI/navigation consumer. The vfs grows `~/blog/` only when posts exist; `StaticContent` gains a hidden-when-empty blog section.

**Tech Stack:** Node script (no deps), vitest with `vi.mock` for the generated module, Playwright e2e.

**Conventions:** Branch `feat/blog-window`. Imperative commits, no trailers. Full gate before each commit: `npm run typecheck && npm run lint && npm run test`.

- [ ] **Step 3: Implement the script**

```js
// scripts/bundle-blog.mjs
/**
 * Bundles data/blog/*.md (frontmatter: title, date) into
 * data/generated/blog.ts, newest first. Zero posts is a valid state —
 * the blog window stays dormant (spec 2026-06-12). README.md is the
 * directory's documentation, never a post. Malformed files are skipped
 * with a warning, never a build failure.
 */
import { mkdirSync, readdirSync, readFileSync, writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";

const root = fileURLToPath(new URL("..", import.meta.url));

/** Parse one post file; returns null when frontmatter is unusable. */
export function parsePost(filename, text) {
  const m = /^---\r?\n([\s\S]*?)\r?\n---\r?\n?([\s\S]*)$/.exec(text);
  if (!m) return null;
  const fields = {};
  for (const line of m[1].split(/\r?\n/)) {
    const kv = /^(\w+)\s*:\s*(.+)$/.exec(line.trim());
    if (kv) fields[kv[1].toLowerCase()] = kv[2].trim();
  }
  if (!fields.title || !fields.date) return null;
  return {
    slug: filename.replace(/\.md$/, ""),
    title: fields.title,
    date: fields.date,
    body: m[2].trim(),
  };
}

function main() {
  let files = [];
  try {
    files = readdirSync(`${root}data/blog`).filter(
      (f) => f.endsWith(".md") && f.toLowerCase() !== "readme.md",
    );
  } catch {
    /* no data/blog directory yet — zero posts */
  }

  const posts = [];
  for (const f of files.sort()) {
    const post = parsePost(f, readFileSync(`${root}data/blog/${f}`, "utf8"));
    if (post) posts.push(post);
    else console.warn(`bundle-blog: skipping ${f} (missing/malformed frontmatter)`);
  }
  posts.sort((a, b) => (a.date < b.date ? 1 : a.date > b.date ? -1 : 0));

  mkdirSync(`${root}data/generated`, { recursive: true });
  writeFileSync(
    `${root}data/generated/blog.ts`,
    `// Generated by scripts/bundle-blog.mjs — do not edit by hand.
export interface BlogPost {
  slug: string;
  title: string;
  date: string; // ISO YYYY-MM-DD
  body: string; // markdown
}

export const blogPosts: BlogPost[] = ${JSON.stringify(posts, null, 2)};
`,
  );
  console.log(`bundle-blog: ${posts.length} post(s)`);
}

if (process.argv[1] === fileURLToPath(import.meta.url)) main();
```

```tsx
// components/content/BlogPosts.tsx
import ReactMarkdown from "react-markdown";
import remarkGfm from "remark-gfm";
import { blogPosts, type BlogPost } from "@/data/generated/blog";

/** Full posts for the plain view; renders nothing while the blog is dormant. */
export function BlogPosts({ posts = blogPosts }: { posts?: BlogPost[] }) {
  if (posts.length === 0) return null;
  return (
    <div className="space-y-10">
      {posts.map((p) => (
        <article key={p.slug} className="space-y-2">
          <h3 className="font-mono text-base font-semibold text-fg">{p.title}</h3>
          <p className="font-mono text-xs text-muted">{p.date}</p>
          <div className="prose-terminal font-sans text-[15px] leading-relaxed text-fg">
            <ReactMarkdown remarkPlugins={[remarkGfm]}>{p.body}</ReactMarkdown>
          </div>
        </article>
      ))}
    </div>
  );
}
```
